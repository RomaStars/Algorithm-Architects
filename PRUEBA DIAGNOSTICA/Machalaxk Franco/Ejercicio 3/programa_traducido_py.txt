#include <stdio.h>
#include <stdlib.h>

#define MAXIMO 100

// Estructura para un Nodo de lista enlazada
tipo_def estructura Nodo {
    entero dato;
    estructura Nodo* siguiente;
} Nodo;

/* 
 * Funcion para crear un nodo en memoria dinamica.
 * Retorna un puntero al nuevo nodo.
 */
Nodo* crear_nodo(entero valor) {
    Nodo* nuevo = (Nodo*)malloc(tamano_de(Nodo));
    si (nuevo == NULL) {
        retornar NULL;
    }
    nuevo->dato = valor;
    nuevo->siguiente = NULL;
    retornar nuevo;
}

entero main(vacio) {
    constante caracter* mensaje = "Iniciando prueba de \"lexer\"\n";
    caracter inicial = 'A';
    sin_signo entero i = 0;
    Nodo* cabeza = crear_nodo(1);
    
    mientras (i < MAXIMO) {
        segun(i % 3) {
            caso 0:
                cabeza->dato += 1;
                romper;
            caso 1:
                // Ignorar este caso
                romper;
            por_defecto:
                cabeza->dato -= 1;
                romper;
        }
        i++;
    }
    
    printf("Resultado final: %d\n", cabeza->dato);
    free(cabeza);
    retornar 0;
}
